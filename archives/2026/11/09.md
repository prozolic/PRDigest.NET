// 修正前: IndexOutOfRangeException
TimeSpan.TryParse("0:0:0.0000000123456789", CultureInfo.InvariantCulture, out _);
// 修正前: IndexOutOfRangeException（1 tick に丸められるべき値）
TimeSpan.TryParse("0:0:0.000000050000000", CultureInfo.InvariantCulture, out _);
// ゼロを1つ追加すると成功（true, TimeSpan.Zero）
TimeSpan.TryParse("0:0:0.00000000123456789", CultureInfo.InvariantCulture, out TimeSpan result);
